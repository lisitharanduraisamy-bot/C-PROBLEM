#include <stdio.h>
#include <string.h>

int main() {
    char username[50];
    int i,len,valid=1;
    printf("Enter username: ");
    scanf("%s",username);
    len=strlen(username);
    if(len<6||len>12){
        valid=0;
    }
    if(username[0]>='0'&&username[0]<='9'){
        valid=0;
    }
    for(i=0;i<len;i++){
        if(!((username[i]>='a'&&username[i]<='z')||
              (username[i]>='0'&&username[i]<='9'))){
            valid=0;
            break;
        }
    }

    if (valid)
        printf("Valid\n");
    else
        printf("Invalid");

    return 0;
}
