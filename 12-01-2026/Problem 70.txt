#include <stdio.h>
#include <string.h>
int main() {
    char s[100];
    gets(s);
    for(int i=0;s[i];i++){
       int count=1;
        if(s[i]== ' ') 
           continue;
        for(int j=i+1;s[j];j++){
            if(s[i]==s[j]){
                count++;
                s[j] = ' ';
            }
        }
        printf("%c %d ",s[i],count);
    }
    return 0;
}
