#include <stdio.h>
#include <string.h>
int main() {
    char s[100],word[20],small[20],large[20];
    int i=0,j=0,len,min=100,max=0;
    gets(s);
    len=strlen(s);
    for(i=0;i<=len;i++){
        if(s[i]!=' '&&s[i]!='\0')
            word[j++]=s[i];
        else{
            word[j]='\0';
            if(j<min){min=j;strcpy(small,word);}
            if(j>max){max=j;strcpy(large,word);}
            j=0;
        }
    }
    printf("%s,%s",large,small);
    return 0;
}
