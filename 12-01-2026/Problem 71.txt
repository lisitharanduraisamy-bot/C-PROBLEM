#include <stdio.h>
#include <string.h>
int main() {
    char a[50],b[50];
    gets(a);
    gets(b);
    if(strlen(a)!=strlen(b)){
        printf("Not Anagram");
        return 0;
    }
    for(int i=0;a[i];i++){
        for(int j=i+1;a[j];j++){
            if(a[i]>a[j]){
                char t=a[i];a[i]=a[j];a[j]=t;
            }
            if(b[i]>b[j]){
                char t=b[i];b[i]=b[j];b[j]=t;
            }
        }
    }
    if(strcmp(a,b)==0)
        printf("Anagram");
    else
        printf("Not Anagram");
    return 0;
}
