#include <stdio.h>
#include <string.h>

int main() {
    char str[200];
    int count =0,i;
    fgets(str,sizeof(str),stdin);
    str[strcspn(str,"\n")]='\0';
    if(strlen(str)==0){
        printf("0");
        return 0;
    }
    count=1;
    for(i=0;str[i]!='\0';i++){
        if(str[i ==' '&&str[i+1]!=' '){
            count++;
        }
    }
    printf("%d",count);
    return 0;
}
