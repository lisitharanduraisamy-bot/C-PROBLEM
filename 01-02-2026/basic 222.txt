#include <stdio.h>

int main() {
    int n,rev=0,temp,r;
    scanf("%d",&n);
    temp=n;
    while(n>0){
        r=n%10;
        rev=rev*10+r;
        n=n/10;
    }
    if(temp==rev)
        printf("Palindrome number\n");
    else
        printf("Not a palindrome\n");

    return 0;
}
