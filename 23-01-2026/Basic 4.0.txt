#include <stdio.h>

int main() {
    int n,target,found=0;
    scanf("%d",&n);

    int arr[n];
    for(int i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    scanf("%d",&target);
    for(int i=0;i<n;i++){
        if(arr[i]==target){
            found=i;
            break;
        }
    }
    if(found!=0)
        printf("Element found at index %d\n",found);
    else
        printf("Element not found\n");
    return 0;
}
