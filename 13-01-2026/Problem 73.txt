#include <stdio.h>
#include <string.h>
int main() {
    char s[100],word[20],longest[20];
    int i=0,j=0,max=0;
    gets(s);
    for(i=0;i<=strlen(s);i++){
        if(s[i]!=' '&&s[i]!='\0')
            word[j++]=s[i];
        else{
            word[j]='\0';
            if(j>max){
                max=j;
                strcpy(longest,word);
            }
            j=0;
        }
    }
    printf("%s",longest);
    return 0;
}
