#include <stdio.h>
#include <string.h>
int main() {
    char s[50];
    gets(s);
    for(int i=0;s[i];i++){
        for(int j=i+1;s[j];j++){
            if(s[i]>s[j]){
                char t=s[i];
                s[i]=s[j];
                s[j]=t;
            }
        }
    }
    printf("%s",s);
    return 0;
}
